  <input type="number" id="entradaAvista" placeholder="Ex: 10000">

  <input type="number" id="parcelaEntrada" placeholder="Ex: 1000">

  <input type="number" id="despesasMensais" placeholder="Ex: 2500">

  <label>Prestação do Financiamento:</label>
  <input type="number" id="prestacaoFinanciamento" placeholder="Ex: 3000">

  <button onclick="simular()">Simular</button>

  <h2>Resultado</h2>
  <table id="resultado">
    <tr>
      <th>Mês</th>
      <th>Total Entradas</th>
      <th>Total Saídas</th>
      <th>Saldo Líquido</th>
    </tr>
  </table>

  <script>
    function simular() {
      const valorImovel = parseFloat(document.getElementById("valorImovel").value) || 0;
      const entradaAvista = parseFloat(document.getElementById("entradaAvista").value) || 0;
      const parcelaEntrada = parseFloat(document.getElementById("parcelaEntrada").value) || 0;
      const despesasMensais = parseFloat(document.getElementById("despesasMensais").value) || 0;
      const prestacaoFinanciamento = parseFloat(document.getElementById("prestacaoFinanciamento").value) || 0;

      const tabela = document.getElementById("resultado");
      tabela.innerHTML = `
        <tr>
          <th>Mês</th>
          <th>Total Entradas</th>
          <th>Total Saídas</th>
          <th>Saldo Líquido</th>
        </tr>
      `;

      // Exemplo: simular 12 meses
      for (let mes = 1; mes <= 12; mes++) {
        const entradas = 9500; // salário fixo de exemplo
        const saidas = despesasMensais + prestacaoFinanciamento + (mes === 1 ? entradaAvista : parcelaEntrada);
        const saldo = entradas - saidas;

        tabela.innerHTML += `
          <tr>
            <td>${mes}</td>
            <td>R$ ${entradas.toFixed(2)}</td>
            <td>R$ ${saidas.toFixed(2)}</td>
            <td class="${saldo >= 0 ? 'positivo' : 'negativo'}">R$ ${saldo.toFixed(2)}</td>
          </tr>
        `;
      }
    }
  </script>
